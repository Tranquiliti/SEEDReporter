#Requires AutoHotkey v2.0
#SingleInstance Force
#MaxThreadsPerHotkey 2

; Just to make sure the coordinates are relative to the Starsector client
CoordMode "Mouse", "Client"

; Button presses and delays assume the SEED Reporter mod is enabled
; Coordinates assume Starsector has fullscreen disabled and is set to 1280x768, 1360x768, or 1366x768 resolution
; Modify the Sleep numbers based on your Starsector FPS and load times
; Note that the first game load will always take longer than subsequent loads
GenerateNewGame() {
	Sleep 800       ; Wait 0.8 seconds
	if not WinActive("ahk_class LWJGL")
		Reload      ; Force-reload script if Starsector was closed during the loop
	Click 1135, 340 ; Click "New Game"
	Sleep 150       ; Wait 0.15 seconds
	Click 965, 200  ; Click "Generate"
	Sleep 150       ; Wait 0.15 seconds
	Send "1"        ; Press "Continue"
	Sleep 150       ; Wait 0.15 seconds
	Send "1"        ; Press option 1
	Sleep 2000      ; Wait 2 seconds for new game to load
	Send "{Escape}" ; Pause game
	Sleep 150       ; Wait 0.15 seconds
	Click 865, 510  ; Click "Exit"
	Sleep 75        ; Wait 0.075 seconds
	Send "{Enter}"  ; Confirm exit
}

; Press the backtick button [`] to activate/deactivate the loop
#HotIf WinActive("ahk_class LWJGL")
`:: {
	static toggle := true
	If toggle := !toggle
		Reload

	Loop {
		GenerateNewGame()
	}
}

newGameGUI := Gui(, "Starsector save auto-generator")
newGameGUI.Add("Text",, "This script assumes the SEED Reporter mod is enabled!")
newGameGUI.Add("Text",, "Press the backtick button to automatically create new Starsector saves.")
newGameGUI.Add("Text",, "Pressing the backtick button again will reload the script, halting the save-generating loop.")
newGameGUI.Add("Text",, "Close this window to fully exit the script.")
newGameGUI.OnEvent("Close", (*) => ExitApp())
newGameGUI.Show()
